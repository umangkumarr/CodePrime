This is a programming version of [Problem 239](https://projecteuler.net/problem=239) from projecteuler.net.

This is clearly a [partial dearrangement](https://mathworld.wolfram.com/PartialDerangement.html) problem. If $a$ is the number of permutation with exactly $k$ prime number found away from their natural position and $b$ is the total number of permutation. Then the answer is $a/b$. 

## Code

```cpp
#include <bits/stdc++.h>

using namespace std;

#define ll long long
const ll mod = 1e9 + 123;

vector<ll> primes;
vector<ll> facts(10000001);

// ------------------ Modular Multiplicative inverse -------------------------

ll modularExponent(ll num, ll pow) {
    num %= mod;
    ll res = 1ll;
    while (pow) {
        ll lb = pow & 1ll;
        if (lb) {
            res = (res * num) % mod;
        }
        pow >>= 1ll;
        num *= num;
        num %= mod;
    }
    return res;
}

ll inv(ll v) { return modularExponent(v, mod - 2ll); }

// -------- Precomputation of all primes less than the upperbound -------------

void preComputation() {
    bitset<10000001> bt;  // indices 0..10^7 inclusive
    for (int i = 2; i <= 10000000; i++) {
        if (!bt[i]) {
            primes.push_back(i);
            for (int j = 2 * i; j <= 10000000; j += i) {
                bt[j] = 1;
            }
        }
    }
    facts[0] = facts[1] = 1;
    for (ll i = 2; i <= 10000000; i++) {
        facts[i] = facts[i - 1] * i;
        facts[i] %= mod;
    }
}

ll fact(ll a) { return facts[a]; }

// ---------------- Partial Dearrangements ----------
// this is not optimised

map<int, int> mp[664581];
// small improvement in performance by avoiding recalculation of calculated shit

ll dearrangements(ll moves, ll dontCare) {
    if(mp[moves][dontCare] > 0) return mp[moves][dontCare];
    
    if (moves <= 0) return fact(dontCare);

    ll ans = (dontCare * dearrangements(moves - 1, dontCare)) % mod;
    if (moves - 2 >= 0) {
        ans += ((moves - 1) * dearrangements(moves - 2, dontCare + 1)) % mod;
    }
    ans %= mod;
    return mp[moves][dontCare] = ans;
}

//------------ Combination ---------------

ll nCr(ll n, ll r) {
    ll ans = (fact(n) * inv(fact(n - r))) % mod;
    ans *= inv(fact(r));
    ans %= mod;
    return ans;
}

// --------------- Something important* -----------

void solve() {
    ll n, k;
    cin >> n >> k;
    ll tot_primes = upper_bound(primes.begin(), primes.end(), n) - primes.begin();

    ll ans = dearrangements(k, n - tot_primes);

    ans *= nCr(tot_primes, k);
    ans %= mod;
    ans *= inv(fact(n));
    ans %= mod;
    cout << ans << "\n";
}

// --------------- main, still not main -----------------

int main() {
    preComputation();

    ll tc = 1;

    while (tc--) {
        solve();
    }
}
```

```java
import java.util.Scanner;

public class Solution {
    static final long MOD = 1_000_000_123L;

    // ------------------ Modular multiplicative inverse -------------------------

    static long modularExponent(long num, long pow) {
        num %= MOD;
        long res = 1;
        while (pow > 0) {
            if ((pow & 1) == 1) res = res * num % MOD;
            pow >>= 1;
            num = num * num % MOD;
        }
        return res;
    }

    static long inv(long v) { return modularExponent(v, MOD - 2); }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();

        // -------- Sieve: number of primes <= n -------------

        boolean[] composite = new boolean[n + 1];
        int totPrimes = 0;
        for (int i = 2; i <= n; i++) {
            if (!composite[i]) {
                totPrimes++;
                for (long j = 2L * i; j <= n; j += i) composite[(int) j] = true;
            }
        }

        long[] fact = new long[n + 1];
        fact[0] = 1;
        for (int i = 1; i <= n; i++) fact[i] = fact[i - 1] * i % MOD;

        // ---------------- Partial derangements ----------
        // D(moves, dontCare) = dontCare!                                   if moves <= 0
        //                    = dontCare * D(moves - 1, dontCare)
        //                      + (moves - 1) * D(moves - 2, dontCare + 1)  otherwise
        // Filled row by row (moves = 0, 1, ..., k); row `moves` only needs
        // dontCare = d0 .. d0 + (k - moves) / 2, the states the recursion would visit.
        int d0 = n - totPrimes;
        long[] prev2 = null, prev1 = null;
        for (int m = 0; m <= k; m++) {
            long[] cur = new long[(k - m) / 2 + 1];
            for (int b = 0; b < cur.length; b++) {
                long dontCare = d0 + b;
                if (m == 0) {
                    cur[b] = fact[d0 + b];
                } else {
                    cur[b] = dontCare * prev1[b] % MOD;
                    if (m >= 2) cur[b] = (cur[b] + (m - 1) * prev2[b + 1]) % MOD;
                }
            }
            prev2 = prev1;
            prev1 = cur;
        }

        // answer = D(k, n - P) * C(P, k) / n!
        long nCr = fact[totPrimes] * inv(fact[totPrimes - k]) % MOD * inv(fact[k]) % MOD;
        long ans = prev1[0] * nCr % MOD * inv(fact[n]) % MOD;
        System.out.println(ans);
    }
}
```

```python
import sys

MOD = 10**9 + 123


def main():
    n, k = map(int, sys.stdin.read().split())

    # -------- Sieve: number of primes <= n -------------
    is_prime = bytearray([1]) * (n + 1)
    is_prime[0] = is_prime[1] = 0
    for i in range(2, int(n**0.5) + 1):
        if is_prime[i]:
            is_prime[i * i :: i] = bytes(len(range(i * i, n + 1, i)))
    tot_primes = sum(is_prime)
    d0 = n - tot_primes

    # factorials mod MOD, kept only at the indices used below
    fact, f, last = {}, 1, 0
    for x in sorted({n, tot_primes, tot_primes - k, k, *range(d0, d0 + k // 2 + 1)}):
        for i in range(last + 1, x + 1):
            f = f * i % MOD
        fact[x] = f
        last = x

    # ---------------- Partial derangements ----------
    # D(moves, dont_care) = dont_care!                                    if moves <= 0
    #                     = dont_care * D(moves - 1, dont_care)
    #                       + (moves - 1) * D(moves - 2, dont_care + 1)   otherwise
    # Filled row by row (moves = 0, 1, ..., k); row `moves` only needs
    # dont_care = d0 .. d0 + (k - moves) // 2, the states the recursion would visit.
    prev2 = prev1 = None
    for m in range(k + 1):
        width = (k - m) // 2 + 1
        if m == 0:
            cur = [fact[d0 + b] for b in range(width)]
        elif m == 1:
            cur = [(d0 + b) * prev1[b] % MOD for b in range(width)]
        else:
            cur = [((d0 + b) * prev1[b] + (m - 1) * prev2[b + 1]) % MOD for b in range(width)]
        prev2, prev1 = prev1, cur

    # answer = D(k, n - P) * C(P, k) / n!
    n_cr = fact[tot_primes] * pow(fact[tot_primes - k], MOD - 2, MOD) % MOD * pow(fact[k], MOD - 2, MOD) % MOD
    print(prev1[0] * n_cr % MOD * pow(fact[n], MOD - 2, MOD) % MOD)


main()
```
