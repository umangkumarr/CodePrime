This is a programming version of [Problem 2](https://projecteuler.net/problem=2) from projecteuler.net.

Find the fibonacci series with the last element just equal to or less then the given $N$. Then sum up all the even valued terms in the series.

## Code

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int tc;
    cin >> tc;

    while (tc--) {
        long long n;
        cin >> n;
        long long a = 0, b = 1, c;
        long long res = 0;
        while (1) {
            c = a + b;
            if (c > n) break;
            if (c % 2 == 0) res += c;
            a = b;
            b = c;
        }
        cout << res << "\n";
    }
}
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int tc = Integer.parseInt(in.readLine().trim());
        StringBuilder out = new StringBuilder();

        while (tc-- > 0) {
            long n = Long.parseLong(in.readLine().trim());
            long a = 0, b = 1;
            long res = 0;
            while (true) {
                long c = a + b;
                if (c > n) break;
                if (c % 2 == 0) res += c;
                a = b;
                b = c;
            }
            out.append(res).append('\n');
        }
        System.out.print(out);
    }
}
```
